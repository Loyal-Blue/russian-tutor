cmake_minimum_required(VERSION 3.22.1)
project(russiantutor_native)
include(FetchContent)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_WASM OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_WEBGPU OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG v0.4.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(llama)
add_library(russiantutor_native SHARED native_tutor.cpp)
target_include_directories(russiantutor_native PRIVATE ${llama_SOURCE_DIR}/include)
target_link_libraries(russiantutor_native PRIVATE llama log android)
target_compile_features(russiantutor_native PRIVATE cxx_std_17)
